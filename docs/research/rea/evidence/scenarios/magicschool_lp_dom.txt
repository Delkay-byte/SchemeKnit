window.addEventListener('load', function () {
  const scrollContainer = document.querySelector('.home_quote_scroll');
  if (!scrollContainer) return;

  const starUrls = [
    'https://cdn.prod.website-files.com/645187265d5e5e386be40629/694051dd34e1e287aff07095_quote-star-blue.svg',
    'https://cdn.prod.website-files.com/645187265d5e5e386be40629/694051dd66334d8de89ef6e8_quote-star-yellow.svg',
    'https://cdn.prod.website-files.com/645187265d5e5e386be40629/694051ddb939a0e51c951550_quote-star-orange.svg',
    'https://cdn.prod.website-files.com/645187265d5e5e386be40629/694051ddccc8d0a20f5d717a_quote-star-purple.svg',
    'https://cdn.prod.website-files.com/645187265d5e5e386be40629/694051dd997d91c4d8ab3ca7_quote-star-teal.svg'
  ];

  const originalCards = Array.from(scrollContainer.querySelectorAll('.quote_scroll')).map(card => {
    const clone = card.cloneNode(true);
    clone.setAttribute('loading', 'eager');
    return clone;
  });

  function waitForImages() {
    const images = scrollContainer.querySelectorAll('img');

    return Promise.all(Array.from(images).map(img => {
      img.setAttribute('loading', 'eager');

      if (img.complete && img.naturalWidth > 0) return Promise.resolve();

      return new Promise(resolve => {
        img.onload = resolve;
        img.onerror = resolve;
        setTimeout(resolve, 3000);
      });
    }));
  }

  function addStar(index) {
    const starDiv = document.createElement('div');
    starDiv.className = 'quote-star-divider';

    const starIndex = index % starUrls.length;

    const topStar = document.createElement('img');
    topStar.src = starUrls[starIndex];
    topStar.alt = '';
    topStar.className = 'star-top';
    topStar.setAttribute('loading', 'eager');

    const bottomStar = document.createElement('img');
    bottomStar.src = starUrls[starIndex];
    bottomStar.alt = '';
    bottomStar.className = 'star-bottom';
    bottomStar.setAttribute('loading', 'eager');

    starDiv.appendChild(topStar);
    starDiv.appendChild(bottomStar);

    return starDiv;
  }

  async function buildScroller() {
    scrollContainer.style.animation = 'none';
    scrollContainer.style.visibility = 'hidden';
    scrollContainer.style.opacity = '0';
    scrollContainer.innerHTML = '';

    originalCards.forEach((card, index) => {
      scrollContainer.appendChild(addStar(index));
      scrollContainer.appendChild(card.cloneNode(true));
    });

    await waitForImages();

    requestAnimationFrame(() => {
      const originalWidth = scrollContainer.scrollWidth;
      const viewportWidth = window.innerWidth;

      if (!originalWidth) {
        scrollContainer.style.visibility = 'visible';
        scrollContainer.style.opacity = '1';
        return;
      }

      const originalChildren = Array.from(scrollContainer.children);
      const timesToClone = Math.ceil((viewportWidth * 2) / originalWidth) + 2;

      for (let i = 0; i < timesToClone; i++) {
        originalChildren.forEach(child => {
          scrollContainer.appendChild(child.cloneNode(true));
        });
      }

      const speed = 50;
      const duration = originalWidth / speed;

      scrollContainer.style.setProperty('--scroll-distance', `-${originalWidth}px`);
      scrollContainer.style.animation = `scrollQuotes ${duration}s linear infinite`;
      scrollContainer.style.visibility = 'visible';
      scrollContainer.style.opacity = '1';
    });
  }

  buildScroller();

  let lastWidth = window.innerWidth;
  let resizeTimer;

  window.addEventListener('resize', function () {
    const currentWidth = window.innerWidth;

    if (currentWidth === lastWidth) return;

    clearTimeout(resizeTimer);

    resizeTimer = setTimeout(() => {
      lastWidth = currentWidth;
      buildScroller();
    }, 250);
  });
});

<!-- Splide Carousel - Logo Carousel (Single Init) -->
<script>
(function(){
  const root = document.querySelector('.logo-carousel-new.splide');
  if (!root) return;

  // stop double init
  if (root.classList.contains('is-initialized')) return;

  const imgs = Array.from(root.querySelectorAll('.splide__slide img'));

  imgs.forEach(img => {
    img.removeAttribute('loading');
    img.decoding = 'async';
    img.fetchPriority = 'high';
  });

  const TARGET_H = 75;
  function sizeSlide(img){
    const slide = img.closest('.splide__slide');
    if (!slide || !img.naturalWidth || !img.naturalHeight) return;
    const w = Math.round(img.naturalWidth * (TARGET_H / img.naturalHeight));
    slide.style.width = w + 'px';
    img.style.height = TARGET_H + 'px';
    img.style.width  = 'auto';
  }

  Promise.all(
    imgs.map(img => (img.complete && img.naturalWidth)
      ? (sizeSlide(img), Promise.resolve())
      : new Promise(res => img.addEventListener('load', () => { sizeSlide(img); res(); }, { once:true }))
    )
  ).then(() => {
    const splide = new Splide(root, {
      type: 'loop',
      arrows: false,
      pagination: false,
      drag: false,
      gap: '2rem',
      autoWidth: true,
      lazyLoad: false,
      autoScroll: { speed: .5, pauseOnHover: false, pauseOnFocus: false },
    });

    splide.mount(window.splide && window.splide.Extensions ? window.splide.Extensions : {});
  });
})();
</script>

<!-- Lottie Animation Tab Control -->
<script>
/**
 * Play Lotties only when their tab is first revealed.
 * - Stops/resets any Lottie inside non-active .w-tab-pane on load
 * - On tab change, plays Lotties in the newly active pane the first time only
 */
(function () {
  function onReady(fn){
    // Run after Webflow has initialized interactions/Lottie
    (window.Webflow = window.Webflow || []).push(fn);
  }

  onReady(function () {
    // Get Lottie library from Webflow or global
    const wfL = (window.Webflow && Webflow.require) ? Webflow.require('lottie') : null;
    const L = (window.lottie) || (wfL && wfL.lottie);
    if (!L || !L.getRegisteredAnimations) return;

    // Map DOM element -> animation for quick lookup
    function buildIndex(){
      const idx = new Map();
      (L.getRegisteredAnimations() || []).forEach(anim => {
        const el = anim.wrapper || anim.container;
        if (el) idx.set(el, anim);
      });
      return idx;
    }

    // Stop & reset all Lotties inside hidden panes
    function resetHiddenPaneLotties() {
      const index = buildIndex();
      document.querySelectorAll('.w-tab-pane:not(.w--tab-active) [data-animation-type="lottie"]').forEach(el => {
        const anim = index.get(el);
        if (anim) {
          anim.stop();
          anim.goToAndStop(0, true);   // reset to frame 0
        }
      });
      // Also mark the currently active pane's Lotties as "not yet played" if needed
      document.querySelectorAll('.w-tab-pane.w--tab-active [data-animation-type="lottie"]').forEach(el => {
        // If they autplayed already, consider them "played"
      });
    }

    function playActivePaneLotties(tabPane) {
      const index = buildIndex();
      tabPane.querySelectorAll('[data-animation-type="lottie"]').forEach(el => {
        const anim = index.get(el);
        if (anim) {
          anim.stop();             // optional: reset to start
          anim.goToAndPlay(0, true);
        }
      });
    }

    // Initial pass: stop hidden panes' animations
    resetHiddenPaneLotties();

    // Watch for tab changes (Webflow toggles .w--current/.w--tab-active)
    document.querySelectorAll('.w-tabs').forEach(tabs => {
      const links = tabs.querySelectorAll('.w-tab-link');
      links.forEach(link => {
        link.addEventListener('click', () => {
          // Wait a tick for Webflow to switch the active pane class
          requestAnimationFrame(() => {
            const active = tabs.querySelector('.w-tab-pane.w--tab-active');
            if (!active) return;
            playActivePaneLotties(active);
          });
        });
      });
    });

    // Safety: if panes are changed via keyboard or script, observe class changes too
    const mo = new MutationObserver(muts => {
      muts.forEach(m => {
        if (m.target.classList && m.target.classList.contains('w-tab-pane') &&
            m.target.classList.contains('w--tab-active')) {
          playActivePaneLotties(m.target);
        }
      });
    });
    document.querySelectorAll('.w-tab-pane').forEach(p => mo.observe(p, { attributes:true, attributeFilter:['class'] }));
  });
})();
</script>

  const menuBtn = (nav && nav.querySelector('.w-nav-button')) || document.querySelector('.w-nav-button');

  let lastY = window.pageYOffset || document.documentElement.scrollTop || 0;
  let ticking = false;

  function getY() {
    return window.pageYOffset || document.documentElement.scrollTop || 0;
  }

  function isMenuOpen() {
    if (!menuBtn) return false;
    return menuBtn.classList.contains('w--open') || menuBtn.getAttribute('aria-expanded') === 'true';
  }

  function showBanner() {
    banner.classList.remove(MENU_OPEN_CLASS);
    banner.classList.remove(HIDDEN_CLASS);
  }

  function hideBanner() {
    banner.classList.remove(MENU_OPEN_CLASS);
    banner.classList.add(HIDDEN_CLASS);
  }

  function hideForMenu() {
    banner.classList.add(MENU_OPEN_CLASS);
    // keep scroll-hidden consistent so when menu closes it doesn't "pop" visible
    banner.classList.add(HIDDEN_CLASS);
  }

  function applyMenuState() {
    if (isMenuOpen()) {
      hideForMenu();
      return;
    }
    // if menu closes and you're at the top, always show
    if (getY() <= TOP_SHOW_PX) showBanner();
  }

  function onScrollRAF() {
    const y = getY();

    // Always show at the very top (prevents ΓÇ£hidden at topΓÇ¥ bug)
    if (y <= TOP_SHOW_PX) {
      if (!isMenuOpen()) showBanner();
      lastY = y;
      return;
    }

    // If menu is open, always hide (no overlap)
    if (isMenuOpen()) {
      hideForMenu();
      lastY = y;
      return;
    }

    const diff = y - lastY;

    // Always update lastY to prevent ΓÇ£stuckΓÇ¥ / sporadic behavior
    if (Math.abs(diff) < DELTA) {
      lastY = y;
      return;
    }

    if (diff > 0) hideBanner();  // scrolling down
    else showBanner();           // scrolling up

    lastY = y;
  }

  // Scroll listener (rAF)
  window.addEventListener('scroll', () => {
    if (ticking) return;
    ticking = true;
    requestAnimationFrame(() => {
      onScrollRAF();
      ticking = false;
    });
  }, { passive: true });

  // Watch menu open/close (class + aria changes)
  if (menuBtn) {
    const mo = new MutationObserver(applyMenuState);
    mo.observe(menuBtn, { attributes: true, attributeFilter: ['class', 'aria-expanded'] });

    // immediate reaction on interaction
    menuBtn.addEventListener('click', () => setTimeout(applyMenuState, 0));
    menuBtn.addEventListener('touchstart', () => setTimeout(applyMenuState, 0), { passive: true });
  }

  // Init
  setTimeout(() => {
    applyMenuState();
    onScrollRAF();
  }, 0);
