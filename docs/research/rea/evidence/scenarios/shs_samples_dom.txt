<script>
(function(){
  var imgs  = Array.prototype.slice.call(document.querySelectorAll('.sample-item img'));
  var box   = document.getElementById('sp-lightbox');
  var boxImg= document.getElementById('sp-lightbox-img');
  var cur   = document.getElementById('sp-current');
  var i     = 0;

  function show(n){
    if(n < 0) n = imgs.length - 1;
    if(n >= imgs.length) n = 0;
    i = n;
    boxImg.src = imgs[i].getAttribute('data-full');
    boxImg.alt = imgs[i].alt;
    cur.textContent = i + 1;
  }
  function open(n){ show(n); box.classList.add('on'); document.body.style.overflow='hidden'; }
  function close(){ box.classList.remove('on'); document.body.style.overflow=''; }

  imgs.forEach(function(im, n){ im.addEventListener('click', function(){ open(n); }); });
  box.querySelector('.sp-close').addEventListener('click', close);
  box.querySelector('.sp-prev').addEventListener('click', function(e){ e.stopPropagation(); show(i-1); });
  box.querySelector('.sp-next').addEventListener('click', function(e){ e.stopPropagation(); show(i+1); });
  box.addEventListener('click', function(e){ if(e.target === box) close(); });
  document.addEventListener('keydown', function(e){
    if(!box.classList.contains('on')) return;
    if(e.key === 'Escape') close();
    if(e.key === 'ArrowLeft') show(i-1);
    if(e.key === 'ArrowRight') show(i+1);
  });
})();
</script>

<script>
(function(){
  var KEY = 'shsp_guide_promo_dismissed';
  var popup = document.getElementById('shsp-promo-popup');
  if (!popup) return;

  function alreadyDismissedToday(){
    try {
      var t = localStorage.getItem(KEY);
      return t && (Date.now() - parseInt(t, 10)) < 86400000; // 24h cooldown
    } catch(e){ return false; }
  }

  window.shspDismissPromo = function(){
    popup.style.display = 'none';
    document.body.style.overflow = '';
    try { localStorage.setItem(KEY, String(Date.now())); } catch(e){}
  };

  if (alreadyDismissedToday()) return;

  var shown = false;
  function showPromo(){
    if (shown || alreadyDismissedToday()) return;
    shown = true;
    popup.style.display = 'flex';
    document.body.style.overflow = 'hidden';
  }

  var timer = setTimeout(showPromo, 20000);
  var trigger = document.getElementById('sample-scroll');
  if (trigger){
    window.addEventListener('scroll', function onScroll(){
      var rect = trigger.getBoundingClientRect();
      if (rect.bottom < window.innerHeight * 0.6){
        window.removeEventListener('scroll', onScroll);
        clearTimeout(timer);
        showPromo();
      }
    }, {passive:true});
  }

  popup.addEventListener('click', function(e){
    if (e.target === popup) shspDismissPromo();
  });
})();
</script>

(function(){
  var sid = sessionStorage.getItem('shsp_sid');
  if (!sid) {
    sid = Math.random().toString(36).slice(2) + Date.now().toString(36);

  }

  // CTA click ΓÇö any link styled as a call-to-action toward the app
  function trackClick(){
    try {
      var data = 'event=click&page=' + encodeURIComponent(location.pathname) + '&sid=' + encodeURIComponent(sid);
      if (navigator.sendBeacon) {
        navigator.sendBeacon('/api/content_click.php', new Blob([data], {type:'application/x-www-form-urlencoded'}));
      } else {
        fetch('/api/content_click.php', {method:'POST', body:data, headers:{'Content-Type':'application/x-www-form-urlencoded'}, keepalive:true}).catch(function(){});
      }
    } catch(e){}
  }
  document.querySelectorAll('.cta-top, .cta-box a, .promo-cta').forEach(function(el){
