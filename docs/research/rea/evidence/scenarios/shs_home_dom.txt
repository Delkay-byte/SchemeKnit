    used: false  },
  // Set only when an admin is actively bypassing an ON maintenance mode ΓÇö
  // never for a regular visitor, since a regular visitor never reaches
  // this far at all when maintenance mode is on (the gate above already
  // exited to maintenance.php for them). Purely so app.js can show an

  maintenanceBypassing: false};

<script>
/* Install prompt: native "Install" button on Android/Chrome; a manual
   "Share -> Add to Home Screen" hint on iPhone Safari (iOS gives no prompt).
   Shown as a centered modal ΓÇö not a bottom bar. Two independent triggers,
   whichever happens first: (1) this is the visitor's 2nd+ visit to the
   site, logged in or out (see bumpVisitCount), or (2) app.js reports a
   successful first paid generation this session via
   window.__shspTriggerInstallPrompt() (see _gatedDownloadDeduct /
   confirmPptGen) ΓÇö this script just owns the actual prompt UI and
   eligibility plumbing (native prompt capture, dismiss cooldown, standalone
   detection).
   Hidden entirely once already installed, or dismissed within 2 days. */
(function(){
  function reportInstall(source){
    try{
      var sid = sessionStorage.getItem('shsp_sid');
      if(!sid){ sid = Math.random().toString(36).slice(2) + Date.now().toString(36); sessionStorage.setItem('shsp_sid', sid); }
      fetch('/tracker.php?action=install&sid=' + encodeURIComponent(sid) + '&source=' + encodeURIComponent(source),
        { method:'GET', credentials:'same-origin' }).catch(function(){});
    }catch(e){}
  }

  var standalone=(window.matchMedia&&window.matchMedia('(display-mode: standalone)').matches)||window.navigator.standalone===true;
  if(standalone){ reportInstall('standalone_launch'); return; }

  function recentlyDismissed(){
    try{ var dis=+localStorage.getItem('shsp_install_dismissed')||0; return dis && (Date.now()-dis)<1728e5; }catch(e){ return false; }
  }

  // Visit counting: one increment per browser SESSION (not per page load),
  // persisted across sessions in localStorage so it survives closing the
  // tab/browser. Works for logged-in and logged-out visitors alike.
  function bumpVisitCount(){
    try{
      if(sessionStorage.getItem('shsp_visit_counted')) return +localStorage.getItem('shsp_visit_count')||0;
      var n=(+localStorage.getItem('shsp_visit_count')||0)+1;
      localStorage.setItem('shsp_visit_count', String(n));
      sessionStorage.setItem('shsp_visit_counted','1');
      return n;
    }catch(e){ return 0; }
  }

  var deferredPrompt=null;
  var isIOS=/iphone|ipad|ipod/i.test(navigator.userAgent)&&!window.MSStream;
  var isSafari=isIOS&&/safari/i.test(navigator.userAgent)&&!/crios|fxios/i.test(navigator.userAgent);
  // Eligible once this is the visitor's 2nd+ visit (logged in or out), OR the
  // instant app.js reports a successful first paid generation ΓÇö whichever
  // happens first. Both are real, independent signals of intent.
  var eligible=(bumpVisitCount()>=2);
  var shownOnce=false;  // don't re-show a second time in the same session after she dismisses it

  function centeredModal(html){
    var overlay=document.createElement('div');
    overlay.id='shsp-install-overlay';
    overlay.style.cssText='position:fixed;inset:0;background:rgba(0,0,0,.55);z-index:99999;display:flex;align-items:center;justify-content:center;padding:1.2rem';
    overlay.onclick=function(e){ if(e.target===overlay) dismiss(); };
    var card=document.createElement('div');
    card.id='shsp-install-banner';
    card.style.cssText='background:#1B6B3A;color:#fff;border-radius:16px;padding:1.5rem 1.3rem;box-shadow:0 20px 60px rgba(0,0,0,.35);max-width:340px;width:100%;text-align:center;font-family:system-ui,sans-serif';
    card.innerHTML=html;
    overlay.appendChild(card);
    document.body.appendChild(overlay);
    return card;
  }
  function dismiss(){
    var el=document.getElementById('shsp-install-overlay');
    if(el) el.remove();
    try{localStorage.setItem('shsp_install_dismissed',String(Date.now()));}catch(e){}
  }
  window.__shspDismissInstall=dismiss;

  function showAndroidPrompt(){
    var card=centeredModal(
      '<img src="/assets/img/icon-192.png" style="width:56px;height:56px;border-radius:14px;margin-bottom:.8rem"/>'
      +'<div style="font-weight:800;font-size:1.08rem;margin-bottom:.5rem">Install SHS Planner</div>'
      +'<div style="opacity:.88;font-size:.86rem;line-height:1.55;margin-bottom:1.1rem">Tap <strong>Install</strong> below to add SHSPlanner to your phone/PC for quick access.</div>'
      +'<button id="shsp-inst-go" style="background:#C8973A;color:#fff;border:none;border-radius:10px;padding:.7rem 1.4rem;font-weight:700;font-size:.9rem;width:100%;margin-bottom:.6rem;cursor:pointer">Install</button>'
      +'<button onclick="__shspDismissInstall()" style="background:transparent;color:#cfe8d8;border:none;font-size:.82rem;text-decoration:underline;cursor:pointer">Not now</button>'
    );
    var go=document.getElementById('shsp-inst-go');
    if(go) go.onclick=function(){
      if(!deferredPrompt) return;
      deferredPrompt.prompt();
      deferredPrompt.userChoice.then(function(){deferredPrompt=null;dismiss();}).catch(function(){deferredPrompt=null;dismiss();});
    };
  }
  function showIOSPrompt(){
    centeredModal(
      '<img src="/assets/img/icon-192.png" style="width:56px;height:56px;border-radius:14px;margin-bottom:.8rem"/>'
      +'<div style="font-weight:800;font-size:1.08rem;margin-bottom:.5rem">Install SHS Planner</div>'
      +'<div style="opacity:.88;font-size:.86rem;line-height:1.55;margin-bottom:1.1rem">Tap the Share icon below, then <strong>Add to Home Screen</strong> for quick, app-like access ΓÇö no browser tabs, just tap and go.</div>'
      +'<button onclick="__shspDismissInstall()" style="background:rgba(255,255,255,.2);color:#fff;border:none;border-radius:10px;padding:.65rem 1.4rem;font-weight:700;font-size:.86rem;cursor:pointer">Got it</button>'
    );
  }

  function maybeShow(){
    if(!eligible||shownOnce) return;
    if(recentlyDismissed()) return; // re-checked live, not just at page-load time
    if(document.getElementById('shsp-install-overlay')) return;
    // BUGFIX (Sept 2026): the Advert/Promo system's Free and Bonus Credit
    // modes are documented (see the Advert admin tab's own description) as
    // replacing ALL of the site's normal pop-ups with the one promo pop-up ΓÇö
    // review request, first-assessment offer, complete-your-profile, AND
    // this install prompt. The first three were correctly wired to
    // _promoBlocksOtherPopups() when that system was built; this one was
    // missed, since it lives in index.php's own separate script block
    // rather than app.js where the other three live. Real-world effect: a
    // teacher on her 2nd visit during an active Free/Bonus promo would see
    // the promo pop-up AND this install card stacked back to back ΓÇö two
    // competing asks instead of the single focused message the promo
    // system exists to show. Checked once, here, since every trigger path
    // above (beforeinstallprompt, the 1200ms post-load timer, and
    // __shspTriggerInstallPrompt after a paid download) already funnels
    // through this one function before anything actually displays.
    //
    // typeof-guarded rather than called directly: this script block loads
    // AFTER app.js (well before this one), so the function is
    // always defined in the normal case ΓÇö but if app.js ever failed to
    // load for an unrelated reason, failing OPEN (still show the install
    // prompt) is safer than a thrown error silently breaking installs
    // entirely for everyone.
    // UPDATE (Oct 2026): the install prompt is no longer switched off during
    // a promo. Instead it waits its turn in app.js's popup coordinator
    // (shspPopupAsk) so it never appears on top of, or right after, another
    // pop-up. If app.js failed to load, fall straight through and show it.
    function doShow(){
      if(shownOnce||recentlyDismissed()) return;
      if(document.getElementById('shsp-install-overlay')) return;
      if(isIOS&&isSafari){ shownOnce=true; showIOSPrompt(); }
      else if(deferredPrompt){ shownOnce=true; showAndroidPrompt(); }
      // else: Android but beforeinstallprompt hasn't fired yet ΓÇö maybeShow()
      // will be called again the moment it does (see listener below), so
      // this just quietly waits rather than showing nothing forever.
    }
    if(typeof shspPopupAsk==='function') shspPopupAsk('install', 6, doShow, { gap:45000 });
    else doShow();
  }

  window.addEventListener('beforeinstallprompt',function(e){ e.preventDefault(); deferredPrompt=e; maybeShow(); });
  window.addEventListener('appinstalled',function(){ reportInstall('appinstalled'); dismiss(); });

  // If this visit already made them eligible (2nd+ visit), check shortly
  // after load ΓÇö covers iOS, which never fires beforeinstallprompt.
  if(eligible) setTimeout(maybeShow, 1200);

  // Secondary trigger: app.js calls this the instant a paid download
  // succeeds for the first time this session ΓÇö still counts as eligibility
  // even on a visitor's very first visit. A tiny delay so it doesn't fight
  // with whatever "download complete" UI is already on screen.
  window.__shspTriggerInstallPrompt=function(){
    if(eligible&&shownOnce) return; // already shown this session
    eligible=true;
    setTimeout(maybeShow, 1200);
  };
})();

</script>
