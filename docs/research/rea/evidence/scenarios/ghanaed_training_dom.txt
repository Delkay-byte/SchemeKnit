<script>
  const slides = Array.from(document.querySelectorAll('.deck .slide'));
  let idx = 0;
  document.getElementById('tot').textContent = slides.length;
  function show(i){
    idx = Math.max(0, Math.min(slides.length - 1, i));
    slides.forEach((s, n) => s.classList.toggle('current', n === idx));
    document.getElementById('cur').textContent = idx + 1;
    document.getElementById('pbar').style.width = ((idx + 1) / slides.length * 100) + '%';
    slides[idx].scrollTop = 0;
  }
  function nextSlide(){ show(idx + 1); }
  function prevSlide(){ show(idx - 1); }
  function startPresent(){
    document.body.classList.add('present');
    show(0);
    if (document.documentElement.requestFullscreen) {
      document.documentElement.requestFullscreen().catch(() => {});
    }
  }
  function exitPresent(){
    document.body.classList.remove('present');
    if (document.fullscreenElement) document.exitFullscreen().catch(() => {});
  }
  document.addEventListener('keydown', e => {
    if (!document.body.classList.contains('present')) {
      if (e.key === 'F5') { e.preventDefault(); startPresent(); }
      return;
    }
    if (['ArrowRight', 'PageDown', ' ', 'Enter'].includes(e.key)) { e.preventDefault(); nextSlide(); }
    else if (['ArrowLeft', 'PageUp', 'Backspace'].includes(e.key)) { e.preventDefault(); prevSlide(); }
    else if (e.key === 'Home') { e.preventDefault(); show(0); }
    else if (e.key === 'End') { e.preventDefault(); show(slides.length - 1); }
    else if (e.key === 'Escape') { exitPresent(); }
  });
  document.addEventListener('fullscreenchange', () => {
    if (!document.fullscreenElement) document.body.classList.remove('present');
  });
</script>
