(function(){

  function trackClick(){
    try {

      if (navigator.sendBeacon) {
        navigator.sendBeacon('/api/content_click.php', new Blob([data], {type:'application/x-www-form-urlencoded'}));
      } else {
        fetch('/api/content_click.php', {method:'POST', body:data, headers:{'Content-Type':'application/x-www-form-urlencoded'}, keepalive:true}).catch(function(){});

    } catch(e){}
